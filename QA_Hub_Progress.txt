--- STEP 5: Refactor /api/logs → Jenkins Console Bridge ---
File: frontend/src/app/api/logs/[runId]/route.ts

Poll Jenkins build console secara periodik dan forward ke SSE:

```typescript
// Pattern: poll /jenkins/job/JOB/BUILD_NUM/logText/progressiveText?start=0
// Parameter "start" adalah byte offset, naik setiap request
// Header X-Text-Size = total size, X-More-Data = true/false (masih running)
let start = 0;
while (true) {
  const res = await fetch(
    `${JENKINS_URL}/jenkins/job/${JOB_NAME}/${buildNum}/logText/progressiveText?start=${start}`,
    { headers: { Authorization: ... } }
  );
  const text = await res.text();
  start = parseInt(res.headers.get("X-Text-Size") || "0");
  const more = res.headers.get("X-More-Data") === "true";
  // kirim lines ke SSE client
  if (!more) break;
  await new Promise(r => setTimeout(r, 1000)); // poll tiap 1 detik
}
```

--- STEP 6: Upload Endpoint ---
File: frontend/src/app/api/scripts/upload/route.ts (BARU)

```typescript
import { writeFile } from "fs/promises";
import path from "path";

export async function POST(request: NextRequest) {
  const formData = await request.formData();
  const file = formData.get("file") as File;
  if (!file?.name.endsWith(".spec.ts"))
    return NextResponse.json({ error: "Only .spec.ts files allowed" }, { status: 400 });
  if (file.size > 1_048_576)
    return NextResponse.json({ error: "Max 1MB" }, { status: 400 });

  const targetDir = process.env.AUTOMATION_DIR || "/data/automation/tests";
  const filePath = path.join(targetDir, file.name);
  await writeFile(filePath, Buffer.from(await file.arrayBuffer()));
  return NextResponse.json({ ok: true, path: file.name });
}
```
